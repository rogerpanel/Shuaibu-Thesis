// 1. Load your 500 m buffer
var buffers = ee.FeatureCollection("projects/disaster-management-474719/assets/WARD_ZONES_BUFFER");

// 2. Landsat 8 Collection 2 L2 for NDVI/EVI
var l8 = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2")
            .filterDate('2024-01-01', '2024-12-31')
            .filterBounds(buffers)
            .map(function(img) {
              return img.select(['SR_B2','SR_B4','SR_B5'])
                        .multiply(0.0000275)
                        .add(-0.2)
                        .copyProperties(img, img.propertyNames());
            })
            .median()
            .clip(buffers);

var ndvi = l8.normalizedDifference(['SR_B5','SR_B4']).rename('NDVI');
var evi = l8.expression(
  '2.5 * ((NIR - RED) / (NIR + 6*RED - 7.5*BLUE + 1))', {
    'NIR': l8.select('SR_B5'),
    'RED': l8.select('SR_B4'),
    'BLUE': l8.select('SR_B2')
}).rename('EVI');

// 3. Night-time Lights (VIIRS)
var ntl = ee.ImageCollection("NOAA/VIIRS/DNB/MONTHLY_V1/VCMCFG")
            .filterDate('2024-01-01', '2024-12-31')
            .mean()
            .select('avg_rad')
            .rename('NTL')
            .clip(buffers);

// 4. Population (WorldPop 100m)
var pop = ee.ImageCollection('WorldPop/GP/100m/pop')
            .filterDate('2020-01-01', '2020-12-31')
            .median()
            .select('population')
            .rename('POP')
            .clip(buffers);

// 5. Built-up % (GHSL)
var builtup = ee.Image("JRC/GHSL/P2016/BUILT_LDSMT_GLOBE_V1")
                 .select('built')
                 .rename('BuiltUP')
                 .clip(buffers);

// 6. SoilGrids 0–5 cm
var soc = ee.Image("projects/soilgrids-isric/soc_mean")
             .select("soc_0-5cm_mean")
             .rename("SOC_0_5cm");
var n   = ee.Image("projects/soilgrids-isric/nitrogen_mean")
             .select("nitrogen_0-5cm_mean")
             .rename("N_0_5cm");
var bd  = ee.Image("projects/soilgrids-isric/bdod_mean")
             .select("bdod_0-5cm_mean")
             .rename("BD_0_5cm");
var cec = ee.Image("projects/soilgrids-isric/cec_mean")
             .select("cec_0-5cm_mean")
             .rename("CEC_0_5cm");
var ph  = ee.Image("projects/soilgrids-isric/phh2o_mean")
             .select("phh2o_0-5cm_mean")
             .rename("pH_0_5cm");

// 7. Stack all variables
var stack = ndvi.addBands([evi, ntl, pop, builtup, soc, n, bd, cec, ph]);

// 8. Compute zonal statistics
var stats = stack.reduceRegions({
  collection: buffers,
  reducer: ee.Reducer.mean(),
  scale: 30
});

// 9. Export
Export.table.toDrive({
  collection: stats,
  description: 'Ward_500m_Extraction',
  fileFormat: 'CSV'
});

print("Extraction ready:", stats);
